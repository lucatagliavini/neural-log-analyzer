# Formato: <pattern ERE> :: <peso>   (peso 1 o 2)
# Righe vuote e righe che iniziano con # sono ignorate.
# L'ordine determina l'indice della feature nel vettore — non riordinare.

# ── Status HTTP ───────────────────────────────────────────────────────────────
errore|errori             :: 2
# Sinonimo diretto di "errore" nel dominio HTTP, e finora SENZA alcuna feature —
# pur essendo usato in 6 esempi labeled su 3 classi (count_status,
# distribute_status, filter_app_errors). Quegli esempi venivano appresi solo
# tramite gli altri token ("totale", "raggruppa", "api"), e una query costruita
# attorno a questa parola non attivava nulla: `richieste fallite` — che è
# letteralmente un esempio di training — non portava alcun tool sopra
# TOOL_THRESHOLD, quindi il bot non rispondeva affatto.
# `gap-report.sh` lo segnalava già («fallimenti — 3 esempi» fra i token non
# coperti) e nessuno aveva agito: l'informazione c'era, il canale funzionava,
# non era stata letta. Stessa famiglia dei checksum obsoleti (c1951e3) e dei test
# fuori dalla suite (NLP-1).
# Prefisso senza \b: copre fallito/fallita/falliti/fallite, fallimento/
# fallimenti, fallisce/falliscono. Peso 2 come `errore|errori`, perché in
# "richieste fallite" è l'UNICA parola che porta il segnale di errore.
fallit|fallim|fallis      :: 2
# \b obbligatorio: senza, "5000" (una soglia in ms) attiva `500` e insegna alla
# rete che un codice di stato può significare "richiesta lenta". Nel corpus
# c'erano 2 esempi slow_requests ("chiamate sopra i 5000ms") che attivavano `500`
# e 1 count_status ("codice errore clv00404") che attivava `404`.
\b500\b                   :: 1
\b400\b                   :: 1
\b404\b                   :: 1
\b503\b                   :: 1
\b5xx\b                   :: 1
\b4xx\b                   :: 1
status|stato              :: 1
http                      :: 1
# "risposte HTTP ricevute", "ricevuti N codici" — solo count_status (1 classe)
rispost[ae]|ricevut       :: 1
# "access log", "accesso da" — max 2 classi (count_status + filter_ip/tail_log)
access\b|accessi\b        :: 1

# ── Tempo / finestre temporali ────────────────────────────────────────────────
# ATTENZIONE — qui NON si usa lo spazio come confine di parola: sia
# query-to-features.sh (`${pattern// /}`) sia build_dataset.py (`.replace(' ','')`)
# RIMUOVONO gli spazi dai pattern, quindi il vecchio `ora |ore |ora$` diventava
# `ora|ore|ora$` e catturava l'interno di **errore** — 21 query del corpus

# che il difetto non si vedeva rileggendo il vocabolario. Dove serve uno spazio si

# "sopra i 2 secondi", "sopra la soglia" — solo slow_requests (1 classe)
sopra\b                   :: 1
# Unità di misura della soglia, come `\bms\b|millisec` sopra. PLURALE OBBLIGATORIO:
# il corpus contiene i due sensi opposti della stessa radice — "quante chiamate al
# second|o|" è una FREQUENZA (traffic_volume, riga 67 del labeled) mentre "sopra i 3
# second|i|" è una SOGLIA (slow_requests, riga 989). Un `second` nudo avrebbe
# codificato quella confusione invece di risolverla; il numero grammaticale la
# discrimina da solo, senza pattern composito.
secondi\b                 :: 1
# Verbi di superamento soglia: "hanno superato i 10 secondi", "oltre i 5 secondi".
# `oltre\b` e non `oltre`: evita "oltremodo" e simili (stessa classe di difetto di
# `per`/`ora` più sopra).

# "tempi di risposta", "response time" — max 4 classi ma peso basso; rinforza service_times e slow_requests
# (NON aggiungere "risposta" da solo: 4 classi, troppo generico — il pattern composito è ok)
tempi\b                   :: 1
# "service times", "SOA service" — solo service_times (1 classe)
service\b                 :: 2

# ── Log applicativi custom specifici (Guidewire nel profilo liquido) ──────────
# UNA sola feature per tutti i log applicativi: normalize-query.sh sostituisce
# qualsiasi "<nome>.log" (e i glob quotati) con il placeholder <LOGFILE>, quindi
# il classificatore impara la *forma* "c'è un nome di logfile qui" e non l'elenco
# dei nomi. Prima c'era l'alternanza esplicita dei 15 nomi di APP_LOG_NAMES: un
# modello legato a un singolo deployment (jgroups esiste solo per app con cache
# distribuita) e fragile — ccCanaliz aveva smesso di funzionare per un problema
# di maiuscole. Sul nodo di produzione i log sono 28, la whitelist ne copriva 16.
#
# Minuscolo perché query-to-features.sh:22-23 e vectorize() fanno lowercase.
# Con le parentesi angolari, non `\blogfile\b`: quest'ultimo matcherebbe anche la
# parola "logfile" digitata dall'utente.
<logfile>                   :: 2
coda|queue\b|batch\b        :: 1
\bdice\b|\bcosa.dice\b      :: 1
# \bguidewire\b — rimosso: normalizzato a <APP> da normalize-query.sh
# api / \bdatabase\b / \bmessaging\b / performance.integr / jgroups /
# plugin|ruleengine|studio / velocity — rimossi: erano nomi di log concreti nel
# vocabolario, ora coperti da <logfile>. Le query che li usavano SENZA ".log"
# ("errori nel jgroups log") sono state riformulate in queries_labeled.txt.

# ── Elenco dei log presenti sul nodo (list_logs) ──────────────────────────────
# Stato del FILESYSTEM ("quali log ci sono sul disco"), non capacità del chatbot
# ("quali log so analizzare" → show_help) né ricerca di contenuto ("in quali log
# c'è X" → search_all_logs). Il discriminante è il predicato di ESISTENZA
# (ci sono / presenti / disponibili / esistono) ancorato al sostantivo "log".
#
# Niente spazi letterali: query-to-features.sh fa `${pattern// /}` e li rimuove.
# `{0,14}` invece di `.*`: ancora il predicato al "log" a distanza di poche
# parole ("log applicativi ci sono", "log di guidewire disponibili") senza
# diventare `che.*analisi`-largo. Verificato: 0 match sui 1026 esempi esistenti.
# Solo list_logs (1 classe) → peso 2, come i discriminatori di show_help/C15.
log.{0,14}(ci.sono|sono.present|present|disponibil|esiston|abbiamo|posso.veder|posso.consultar|ho.a.disposizion|trovo) :: 2
# Forma imperativa/nominale: "lista dei log", "elenco log del nodo 5".
# NON `(lista|elenco).*log`: quella forma cattura 6 esempi esistenti (righe 403,
# 411, 412, 413, 476, 477 — tail_log e filter_app_errors). Con l'ancoraggio
# {0,14} + le sole forme "elenca i/tutti i log" i match diventano 0. Verificato.
# Solo list_logs (1 classe) → peso 2.
lista.{0,14}log|elenco.{0,14}log|elenca.i.log|elenca.tutti.i.log|elenca.i.file.di.log :: 2

# ── <PATTERN>: stringa quotata NON glob-like (QUOTE-1, 2026-08-19) ────────────
# Simmetrico a <logfile> (sopra): normalize-query.sh distingue una stringa
# quotata glob-like ("*-cc.log", nome parziale di log) da qualsiasi altra
# stringa quotata (stringa di ricerca) e sostituisce quest'ultima con questo
# placeholder. Senza, il classificatore non può imparare il confine fra
# "cerca errori nel server.log" (nessun pattern) e 'cerca "NPE" nel
# server.log" (pattern esplicito): il contenuto letterale fra virgolette non è
# nel vocabolario e varia per ogni query.
# Peso 2, come <logfile>: singolo segnale forte, non un token qualsiasi.
<pattern> :: 2
